⑥ 動かす

OrbitSound.cs ― モノを動かすと、音も動く

音の位置 ＝ GameObject の位置。transform.position を動かせば、音も一緒に動く。

using UnityEngine;

// 自分のまわりを、半径 1m の円でゆっくり回す
public class OrbitSound : MonoBehaviour
{
    public float radius = 1.0f;   // 円の半径（m）
    public float speed  = 30f;    // 1秒に進む角度（度）
    public float height = 0f;     // 耳の高さからの上下（m）

    float angle = 0f;

    void Update()
    {
        angle += speed * Time.deltaTime;                        // 角度を進める
        float rad = angle * Mathf.Deg2Rad;                      // 度 → ラジアン
        float y = Camera.main.transform.position.y + height;    // カメラ（耳）の高さ
        transform.position = new Vector3(Mathf.Cos(rad) * radius, y, Mathf.Sin(rad) * radius);
    }
}

⑧ スクリプト

TapSound.cs ― タップしたら 1回鳴らす

using UnityEngine;
using UnityEngine.InputSystem;   // 新しい Input System

// タップ（Mac ではクリック）すると、音を1回鳴らす
public class TapSound : MonoBehaviour
{
    public AudioClip tapClip;          // タップで鳴らす短い音
    private AudioSource audioSource;

    void Start()
    {
        audioSource = GetComponent<AudioSource>();
    }

    void Update()
    {
        // 画面を押した瞬間
        if (Pointer.current.press.wasPressedThisFrame)
        {
            // 押した点から、奥へ線（Ray）を飛ばす
            Vector2 point = Pointer.current.position.ReadValue();
            Ray ray = Camera.main.ScreenPointToRay(point);

            // 線が当たったのが、このモノなら
            if (Physics.Raycast(ray, out RaycastHit hit) && hit.transform.IsChildOf(transform))
            {
                audioSource.PlayOneShot(tapClip);   // 1回鳴らす
            }
        }
    }
}
1
Pointer.current.press押した瞬間（タップもクリックも）
2
ScreenPointToRay → Raycast線を飛ばして、当たったか
3
IsChildOf(transform)当たったのが自分か
4
PlayOneShot(tapClip)1回鳴らす（ループは止めない）
春学期の Input.GetKeyDown は、このプロジェクトでは使えない
